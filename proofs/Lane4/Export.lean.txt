namespace Lane4.Export
open Lane4.Seq

def showQ (q : ℚ) : String := "\"" ++ toString q.num ++ "/" ++ toString q.den ++ "\""
def showV (v : Vec) : String := "[" ++ ", ".intercalate (v.map showQ) ++ "]"
def showM (m : List Vec) : String := "[" ++ ", ".intercalate (m.map showV) ++ "]"
def showMask : Mask → String
  | Mask.all => "\"all\""
  | Mask.causal => "\"causal\""
  | Mask.strict => "\"strict\""
def showH (h : Head) : String :=
  "{\"A\": " ++ showM h.A ++ ", \"V\": " ++ showM h.V ++ ", \"mask\": " ++ showMask h.mask ++ "}"
def showB (b : Block) : String :=
  "{\"heads\": [" ++ ", ".intercalate (b.heads.map showH) ++ "], \"W1\": " ++ showM b.W1 ++
  ", \"b1\": " ++ showV b.b1 ++ ", \"W2\": " ++ showM b.W2 ++ ", \"b2\": " ++ showV b.b2 ++ "}"
def showN (n : TNet) : String :=
  "{\"embed\": " ++ showM n.embed ++ ", \"blocks\": [" ++ ", ".intercalate (n.blocks.map showB) ++
  "], \"outW\": " ++ showV n.outW ++ ", \"outB\": " ++ showQ n.outB ++ "}"

end Lane4.Export

#eval IO.println ("NETJSON " ++ Lane4.Export.showN Lane4.Case.net)
